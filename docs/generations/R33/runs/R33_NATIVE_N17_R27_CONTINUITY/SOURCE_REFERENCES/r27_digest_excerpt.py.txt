@dataclass
class R27State:
    format:str
    r26_sha256:str
    base_state:Any
    category_system:Dict[str,Any]
    affordance_model:Any
    speech_motif_decoder:Any
    abstraction_policy:Dict[str,Any]
    semantic_generator:Any
    visual_debate_policy:Dict[str,Any]
    architecture:Dict[str,Any]
    evidence:Dict[str,Any]
    self_revision_history:List[Dict[str,Any]]
    development_step:int
    newborn_restarts:int=0
    def digest(self):
        h=hashlib.sha256();h.update(self.format.encode());h.update(self.r26_sha256.encode());h.update(str(self.development_step).encode())
        h.update(self.base_state.digest().encode())
        for obj in (self.category_system.get('category_head'),self.category_system.get('entity_head'),self.affordance_model):
            net=getattr(obj,'net',None)
            if net is not None:
                for k,v in sorted(net.state_dict().items()):h.update(k.encode());h.update(v.detach().cpu().numpy().tobytes())
        if hasattr(self.semantic_generator,'specialists'):
            h.update(str(sorted(self.semantic_generator.specialists)).encode())
        h.update(json.dumps(self.architecture,sort_keys=True,default=str).encode())
        h.update(json.dumps(self.evidence,sort_keys=True,default=str).encode())
        return h.hexdigest()
