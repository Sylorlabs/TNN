fn gcd(a:i64,b:i64)i64 {
  let x:i64=a;
  let y:i64=b;
  while(y!=0){
    let t:i64=x%y;
    x=y;
    y=t;
  }
  return x;
}
fn main()void {
  let a:i64=15;
  let b:i64=25;
  let g:i64=gcd(a,b);
  let l:i64=a*b/g;
  _zag_print(_zag_i64_to_str(l));
  _zag_print("\n");
}
