// How it works: count symbol frequencies over the input; build the code
// tree by repeatedly joining the two lightest nodes, breaking ties by

fn parse_i64(s:[]u8)i64 {
    let neg:i64=0;
    let i:i64=0;
    if(s.len>0){
        if(s[0]==45){neg=1;i=1;}
    }
    let v:i64=0;
    while(i<s.len){
        let c:u8=s[i];
        if(c>=48){
            if(c<=57){v=v*10+((c-48) as i64);}
        }
        i=i+1;
    }
    if(neg==1){v=0-v;}
    return v;
}

        if(c<=57){return (c-48) as i64;}

fn s_eq(a:[]u8, b:[]u8)i64 {
    if(a.len!=b.len){return 0;}

fn s_cmp(a:[]u8, b:[]u8)i64 {
    let n:i64=a.len;
    if(b.len<n){n=b.len;}
    let i:i64=0;
    while(i<n){
        if(a[i]!=b[i]){
            if(a[i]<b[i]){return -1;}
            return 1;
        }
        i=i+1;
    }
    if(a.len==b.len){return 0;}
    if(a.len<b.len){return -1;}
    return 1;
}

        let want:i64=cap-total;
        if(want>65536){want=65536;}
        let r:i64=_zag_raw_syscall(0, fd, (mem as i64)+total, want, 0, 0, 0);
        if(r<=0){break;}
        total=total+r;
    }
    let c:i64=_zag_raw_syscall(3, fd, 0, 0, 0, 0, 0);
    return total;
}

fn join_path(a:[]u8, b:[]u8)[]u8 {
    let p:*u8=_zag_malloc(a.len+1+b.len) as *u8;
    let s:[]u8=p[0..a.len+1+b.len];
    s_copy(s, a);
    s[a.len]=47;
    s_copy(s[a.len+1..], b);
    return s;
}

// ---- table storage ----
// per table: ncols, nrows, colnames(16*64), colint(16), cells(nrows*16*64), ival(nrows*16*8)
fn t_ncols(th:[]u8)i64 { return get64(th,0); }
fn t_nrows(th:[]u8)i64 { return get64(th,8); }

fn load_table(dbdir:[]u8, fname:[]u8, th:[]u8)void {
    let path:[]u8=join_path(dbdir, fname);
    let mp:*u8=_zag_malloc(65536) as *u8;
    let n:i64=read_all(path, mp, 65536);
    let buf:[]u8=mp[0..n];
    // count lines and max fields
    let nlines:i64=0;
    let i:i64=0;
    while(i<n){
        if(buf[i]==10){nlines=nlines+1;}
        i=i+1;
    }
    // split into lines: store line start/end in temp arrays
    let lp:*u8=_zag_malloc(nlines*16+16) as *u8;
    let LS:[]u8=lp[0..nlines*16+16];
    let lc:i64=0;
    let ls:i64=0;
    let j:i64=0;
    while(j<=n){
        let eol:i64=0;
        if(j==n){eol=1;}
        if(eol==0){
            if(buf[j]==10){eol=1;}
        }
        if(eol==1){
            let ln:i64=j-ls;
            if(ln>0){
                if(buf[j-1]==13){ln=ln-1;}
            }
            if(ln>0){
                put64(LS,lc*16,ls);
                put64(LS,lc*16+8,ln);
                lc=lc+1;
            }
            ls=j+1;
        }
        j=j+1;
    }
    // header: count fields
    let h0:i64=get64(LS,0);
    let hl:i64=get64(LS,8);
    let ncols:i64=1;
    let k:i64=0;
    while(k<hl){
        if(buf[h0+k]==44){ncols=ncols+1;}
        k=k+1;
    }
    let nrows:i64=lc-1;
    put64(th,0,ncols);
    put64(th,8,nrows);
    // colnames
    let cnp:*u8=_zag_malloc(16*64) as *u8;
    let CN:[]u8=cnp[0..16*64];
    put64(th,16,cnp as i64);
    let cip:*u8=_zag_malloc(16) as *u8;
    let CI:[]u8=cip[0..16];
    put64(th,24,cip as i64);
    let cp:*u8=_zag_malloc((nrows*16*64) as i64) as *u8;
    let CE:[]u8=cp[0..(nrows*16*64) as i64];
    put64(th,32,cp as i64);
    let vp:*u8=_zag_malloc((nrows*16*8) as i64) as *u8;
    let VA:[]u8=vp[0..(nrows*16*8) as i64];
    put64(th,40,vp as i64);
    // header names
    let f:i64=0;
    let fs:i64=h0;
    let kk:i64=0;
    while(kk<=hl){
        let comma:i64=0;
        if(kk==hl){comma=1;}
        if(comma==0){
            if(buf[h0+kk]==44){comma=1;}
        }
        if(comma==1){
            let fl:i64=h0+kk-fs;
            let dst:[]u8=CN[f*64..f*64+64];
            let c2:i64=0;
            while(c2<64){dst[c2]=0;c2=c2+1;}
            s_copy(dst, buf[fs..fs+fl]);
            put64(CN,f*64+56,fl);
            fs=h0+kk+1;
            f=f+1;
        }
        kk=kk+1;
    }
    // rows
    let r:i64=0;
    while(r<nrows){
        let l0:i64=get64(LS,(r+1)*16);
        let ll:i64=get64(LS,(r+1)*16+8);
        let cf:i64=0;
        let cs:i64=l0;
        let kk2:i64=0;
        while(kk2<=ll){
            let comma2:i64=0;
            if(kk2==ll){comma2=1;}
            if(comma2==0){
                if(buf[l0+kk2]==44){comma2=1;}
            }
            if(comma2==1){
                let fl2:i64=l0+kk2-cs;
                let cell:[]u8=buf[cs..cs+fl2];
                let dst2:[]u8=CE[(r*16+cf)*64..(r*16+cf)*64+64];
                let c3:i64=0;
                while(c3<64){dst2[c3]=0;c3=c3+1;}
                s_copy(dst2, cell);
                put64(dst2,56,fl2);
                // int value
                let iv:i64=parse_i64_full(cell);
                put64(VA,(r*16+cf)*8,iv);
                cs=l0+kk2+1;
                cf=cf+1;
            }
            kk2=kk2+1;
        }
        r=r+1;
    }
    // column int detection
    let c4:i64=0;
    while(c4<ncols){
        let allint:i64=1;
        let r2:i64=0;
        while(r2<nrows){
            if(get64(VA,(r2*16+c4)*8)==4611686018427387904){allint=0;}
            r2=r2+1;
        }
        CI[c4]=allint as u8;
        c4=c4+1;
    }
    return;
}

fn col_name(th:[]u8, c:i64)[]u8 {
    let CN:[]u8=(get64(th,16) as *u8)[0..16*64];
    let fl:i64=get64(CN,c*64+56);
    return CN[c*64..c*64+fl];
}

fn cell_str(th:[]u8, r:i64, c:i64)[]u8 {
    let CE:[]u8=(get64(th,32) as *u8)[0..get64(th,8)*16*64];
    let fl:i64=get64(CE,(r*16+c)*64+56);
    return CE[(r*16+c)*64..(r*16+c)*64+fl];
}

fn cell_int(th:[]u8, r:i64, c:i64)i64 {
    let VA:[]u8=(get64(th,40) as *u8)[0..get64(th,8)*16*8];
    return get64(VA,(r*16+c)*8);
}

fn col_is_int(th:[]u8, c:i64)i64 {
    let CI:[]u8=(get64(th,24) as *u8)[0..16];
    return CI[c] as i64;
}

fn find_col(th:[]u8, name:[]u8)i64 {
    let n:i64=t_ncols(th);
    let c:i64=0;
    while(c<n){
        if(s_eq(col_name(th,c),name)==1){return c;}
        c=c+1;
    }
    return -1;
}

// ---- tokenizer ----
// token kinds stored in parallel arrays: kind(0=word,1=quoted,2=punct), text
fn tokenize(q:[]u8, TK:[]u8, TT:[]u8) i64 {
    // TK: ntok*16 (kind@0, start@8) ; TT: ntok*16 (len@0)
    let ntok:i64=0;
    let i:i64=0;
    while(i<q.len){
        let c:u8=q[i];
        if(c==32){i=i+1;}
        else {
            if(c==9){i=i+1;}
            else {
                if(c==39){
                    // quoted
                    let s:i64=i+1;
                    let e:i64=s;
                    while(e<q.len){
                        if(q[e]==39){break;}
                        e=e+1;
                    }
                    put64(TK,ntok*16,1);
                    put64(TK,ntok*16+8,s);
                    put64(TT,ntok*16,e-s);
                    ntok=ntok+1;
                    i=e+1;
                } else {
                    if(c==44){put64(TK,ntok*16,2);put64(TK,ntok*16+8,i);put64(TT,ntok*16,1);ntok=ntok+1;i=i+1;}
                    else {
                        if(c==61){put64(TK,ntok*16,2);put64(TK,ntok*16+8,i);put64(TT,ntok*16,1);ntok=ntok+1;i=i+1;}
                        else {
                            if(c==60){put64(TK,ntok*16,2);put64(TK,ntok*16+8,i);put64(TT,ntok*16,1);ntok=ntok+1;i=i+1;}
                            else {
                                if(c==62){put64(TK,ntok*16,2);put64(TK,ntok*16+8,i);put64(TT,ntok*16,1);ntok=ntok+1;i=i+1;}
                                else {
                                    if(c==33){put64(TK,ntok*16,2);put64(TK,ntok*16+8,i);put64(TT,ntok*16,1);ntok=ntok+1;i=i+1;}
                                    else {
                                        if(c==59){i=i+1;}
                                        else {
                                            let s2:i64=i;
                                            while(i<q.len){
                                                let d:u8=q[i];
                                                if(d==32){break;}
                                                if(d==9){break;}
                                                if(d==39){break;}
                                                if(d==44){break;}
                                                if(d==61){break;}
                                                if(d==60){break;}
                                                if(d==62){break;}
                                                if(d==33){break;}
                                                if(d==59){break;}
                                                i=i+1;
                                            }
                                            put64(TK,ntok*16,0);
                                            put64(TK,ntok*16+8,s2);
                                            put64(TT,ntok*16,i-s2);
                                            ntok=ntok+1;
                                        }
                                    }
                                }
                            }
                        }
                    }
                }
            }
        }
    }
    return ntok;
}

fn tok_text(q:[]u8, TK:[]u8, TT:[]u8, t:i64)[]u8 {
    let s:i64=get64(TK,t*16+8);
    let l:i64=get64(TT,t*16);
    return q[s..s+l];
}

fn tok_kind(TK:[]u8, t:i64)i64 {
    return get64(TK,t*16);
}

fn kw_upper_eq(t:[]u8, lit:[]u8)i64 {
    // case-insensitive compare against uppercase lit
    if(t.len!=lit.len){return 0;}
    let i:i64=0;
    while(i<t.len){
        let c:u8=t[i];
        if(c>=97){
            if(c<=122){c=c-32;}
        }
        if(c!=lit[i]){return 0;}
        i=i+1;
    }
    return 1;
}

fn main()void {
    let dbdir:[]u8=_zag_arg(1);
    let query:[]u8=_zag_arg(2);
    // load tables
    let thp:*u8=_zag_malloc(2*48) as *u8;
    let TH:[]u8=thp[0..96];
    let t1:[]u8=TH[0..48];
    let t2:[]u8=TH[48..96];
    load_table(dbdir, "employees.csv", t1);
    load_table(dbdir, "departments.csv", t2);
    let tnames0:[]u8="employees";
    let tnames1:[]u8="departments";
    // tokenize
    let tkp:*u8=_zag_malloc(256*16) as *u8;
    let ttp:*u8=_zag_malloc(256*16) as *u8;
    let TK:[]u8=tkp[0..256*16];
    let TT:[]u8=ttp[0..256*16];
    let ntok:i64=tokenize(query, TK, TT);
    // parse: SELECT collist FROM t [JOIN u ON cond] [WHERE cond AND..] [ORDER BY ref]
    let p:i64=0;
    // expect SELECT
    p=p+1;
    // select list
    let selp:*u8=_zag_malloc(32*64) as *u8;
    let SEL:[]u8=selp[0..32*64];
    let nsel:i64=0;
    let sel_star:i64=0;
    while(p<ntok){
        if(tok_kind(TK,p)==2){
            // comma
            p=p+1;
        } else {
            let w:[]u8=tok_text(query,TK,TT,p);
            if(kw_upper_eq(w,"FROM")==1){break;}
            if(s_eq(w,"*")==1){sel_star=1;}
            put64(SEL,nsel*64,w.len);
            s_copy(SEL[nsel*64+8..nsel*64+64], w);
            nsel=nsel+1;
            p=p+1;
        }
    }
    // expect FROM
    p=p+1;
    let tfrom:[]u8=tok_text(query,TK,TT,p);
    p=p+1;
    let has_join:i64=0;
    let tjoin:[]u8="";
    // ON condition: left ref, op, right ref
    let on_l:*u8=_zag_malloc(64) as *u8;
    let on_r:*u8=_zag_malloc(64) as *u8;
    let ONL:[]u8=on_l[0..64];
    let ONR:[]u8=on_r[0..64];
    let on_ll:i64=0;
    let on_rl:i64=0;
    if(p<ntok){
        let w2:[]u8=tok_text(query,TK,TT,p);
        if(kw_upper_eq(w2,"JOIN")==1){
            has_join=1;
            p=p+1;
            tjoin=tok_text(query,TK,TT,p);
            p=p+1;
            p=p+1; // ON
            // left ref
            let lr:[]u8=tok_text(query,TK,TT,p);
            s_copy(ONL, lr);
            on_ll=lr.len;
            p=p+1;
            // op tokens (= or two-char)
            let op1:[]u8=tok_text(query,TK,TT,p);
            p=p+1;
            // right ref
            let rr:[]u8=tok_text(query,TK,TT,p);
            s_copy(ONR, rr);
            on_rl=rr.len;
            p=p+1;
        }
    }
    // WHERE conds: arrays of (left ref, op, lit text, lit kind)
    let wp:*u8=_zag_malloc(16*192) as *u8;
    let WC:[]u8=wp[0..16*192];
    let nwc:i64=0;
    if(p<ntok){
        let w3:[]u8=tok_text(query,TK,TT,p);
        if(kw_upper_eq(w3,"WHERE")==1){
            p=p+1;
            while(p<ntok){
                let w4:[]u8=tok_text(query,TK,TT,p);
                if(kw_upper_eq(w4,"ORDER")==1){break;}
                if(kw_upper_eq(w4,"AND")==1){p=p+1;}
                else {
                    // left ref
                    let lr2:[]u8=tok_text(query,TK,TT,p);
                    s_copy(WC[nwc*192..], lr2);
                    put64(WC,nwc*192+64,lr2.len);
                    p=p+1;
                    // op: may be 1 or 2 punct tokens
                    let oo:[]u8=tok_text(query,TK,TT,p);
                    let ob:[]u8=WC[nwc*192+72..nwc*192+80];
                    s_copy(ob, oo);
                    let opl:i64=oo.len;
                    p=p+1;
                    if(p<ntok){
                        if(tok_kind(TK,p)==2){
                            let oo2:[]u8=tok_text(query,TK,TT,p);
                            // combine if oo is <,>,! and oo2 is =
                            if(oo.len==1){
                                if(oo[0]==60){if(oo2[0]==61){ob[1]=61;opl=2;p=p+1;}}
                                if(oo[0]==62){if(oo2[0]==61){ob[1]=61;opl=2;p=p+1;}}
                                if(oo[0]==33){if(oo2[0]==61){ob[1]=61;opl=2;p=p+1;}}
                            }
                        }
                    }
                    put64(WC,nwc*192+80,opl);
                    // literal
                    let lit:[]u8=tok_text(query,TK,TT,p);
                    let lk:i64=tok_kind(TK,p);
                    s_copy(WC[nwc*192+88..], lit);
                    put64(WC,nwc*192+152,lit.len);
                    put64(WC,nwc*192+160,lk);
                    p=p+1;
                    nwc=nwc+1;
                }
            }
        }
    }
    // ORDER BY
    let has_order:i64=0;
    let obp:*u8=_zag_malloc(64) as *u8;
    let OB:[]u8=obp[0..64];
    let ob_l:i64=0;
    if(p<ntok){
        let w5:[]u8=tok_text(query,TK,TT,p);
        if(kw_upper_eq(w5,"ORDER")==1){
            p=p+2; // ORDER BY
            let orf:[]u8=tok_text(query,TK,TT,p);
            s_copy(OB, orf);
            ob_l=orf.len;
            has_order=1;
            p=p+1;
        }
    }
    // resolve table handles
    let TA:[]u8=t1;
    let TB:[]u8=t1;
    if(s_eq(tfrom,"departments")==1){TA=t2;}
    if(has_join==1){
        TB=t2;
        if(s_eq(tjoin,"employees")==1){TB=t1;}
        // TA stays tfrom's
    }
    // helper: resolve ref text -> (table sel, col)
    // collected select items: for each, (table handle idx 0/1, col idx, header text)
    let sip:*u8=_zag_malloc(32*24) as *u8;
    let SI:[]u8=sip[0..32*24];
    let nsi:i64=0;
    if(sel_star==1){
        let c:i64=0;
        while(c<t_ncols(TA)){
            put64(SI,nsi*24,0);
            put64(SI,nsi*24+8,c);
            let cn:[]u8=col_name(TA,c);
            put64(SI,nsi*24+16,cn.len);
            s_copy(SEL[nsi*64+8..], cn);
            nsi=nsi+1;
            c=c+1;
        }
    } else {
        let q2:i64=0;
        while(q2<nsel){
            let sl:i64=get64(SEL,q2*64);
            let ref:[]u8=SEL[q2*64+8..q2*64+8+sl];
            // split table.col
            let dot:i64=-1;
            let d:i64=0;
            while(d<ref.len){
                if(ref[d]==46){dot=d;}
                d=d+1;
            }
            let tn:[]u8="";
            let cn2:[]u8=ref;
            if(dot>=0){
                tn=ref[0..dot];
                cn2=ref[dot+1..ref.len];
            }
            let ti:i64=0; // 0=TA,1=TB
            let ci:i64=-1;
            if(tn.len>0){
                if(s_eq(tn,tnames0)==1){
                    if(s_eq(tfrom,tnames0)==1){ti=0;}
                    else {ti=1;}
                }
                if(s_eq(tn,tnames1)==1){
                    if(s_eq(tfrom,tnames1)==1){ti=0;}
                    else {ti=1;}
                }
            } else {
                ci=find_col(TA,cn2);
                if(ci<0){
                    if(has_join==1){
                        ci=find_col(TB,cn2);
                        ti=1;
                    }
                }
            }
            if(tn.len>0){
                if(ti==0){ci=find_col(TA,cn2);}
                else {ci=find_col(TB,cn2);}
            }
            put64(SI,q2*24,ti);
            put64(SI,q2*24+8,ci);
            put64(SI,q2*24+16,sl);
            nsi=nsi+1;
            q2=q2+1;
        }
    }
    // resolve ON refs
    let on_lc:i64=-1;
    let on_lt:i64=0;
    let on_rc:i64=-1;
    let on_rt:i64=0;
    if(has_join==1){
        let ol:[]u8=ONL[0..on_ll];
        let orr:[]u8=ONR[0..on_rl];
        // split each on dot
        let d1:i64=-1;
        let d2:i64=-1;
        let x:i64=0;
        while(x<ol.len){if(ol[x]==46){d1=x;}x=x+1;}
        x=0;
        while(x<orr.len){if(orr[x]==46){d2=x;}x=x+1;}
        let olt:[]u8=ol[0..d1];
        let olc:[]u8=ol[d1+1..ol.len];
        let ort2:[]u8=orr[0..d2];
        let orc:[]u8=orr[d2+1..orr.len];
        if(s_eq(olt,tfrom)==1){on_lt=0;on_lc=find_col(TA,olc);}
        else {on_lt=1;on_lc=find_col(TB,olc);}
        if(s_eq(ort2,tfrom)==1){on_rt=0;on_rc=find_col(TA,orc);}
        else {on_rt=1;on_rc=find_col(TB,orc);}
    }
    // resolve WHERE refs
    let wci:i64=0;
    while(wci<nwc){
        let rl:i64=get64(WC,wci*192+64);
        let ref2:[]u8=WC[wci*192..wci*192+rl];
        let dot2:i64=-1;
        let d3:i64=0;
        while(d3<ref2.len){if(ref2[d3]==46){dot2=d3;}d3=d3+1;}
        let tn3:[]u8="";
        let cn3:[]u8=ref2;
        if(dot2>=0){tn3=ref2[0..dot2];cn3=ref2[dot2+1..ref2.len];}
        let ti3:i64=0;
        let ci3:i64=-1;
        if(tn3.len>0){
            if(s_eq(tn3,tfrom)==1){ci3=find_col(TA,cn3);}
            else {ti3=1;ci3=find_col(TB,cn3);}
        } else {
            ci3=find_col(TA,cn3);
            if(ci3<0){
                if(has_join==1){ti3=1;ci3=find_col(TB,cn3);}
            }
        }
        put64(WC,wci*192+168,ti3);
        put64(WC,wci*192+176,ci3);
        wci=wci+1;
    }
    // resolve ORDER BY ref
    let ob_t:i64=0;
    let ob_c:i64=-1;
    if(has_order==1){
        let orf2:[]u8=OB[0..ob_l];
        let dot3:i64=-1;
        let d4:i64=0;
        while(d4<orf2.len){if(orf2[d4]==46){dot3=d4;}d4=d4+1;}
        let tn4:[]u8="";
        let cn4:[]u8=orf2;
        if(dot3>=0){tn4=orf2[0..dot3];cn4=orf2[dot3+1..orf2.len];}
        if(tn4.len>0){
            if(s_eq(tn4,tfrom)==1){ob_c=find_col(TA,cn4);}
            else {ob_t=1;ob_c=find_col(TB,cn4);}
        } else {
            ob_c=find_col(TA,cn4);
            if(ob_c<0){
                if(has_join==1){ob_t=1;ob_c=find_col(TB,cn4);}
            }
        }
    }
    // execute: row pairs
    let n1:i64=t_nrows(TA);
    let n2:i64=1;
    if(has_join==1){n2=t_nrows(TB);}
    let rpp:*u8=_zag_malloc((n1*n2*16+16) as i64) as *u8;
    let RP:[]u8=rpp[0..(n1*n2*16+16) as i64];
    let nrp:i64=0;
    let r1:i64=0;
    while(r1<n1){
        let r2:i64=0;
        while(r2<n2){
            let pass_on:i64=1;
            if(has_join==1){

// split full child y=p.children[i] (2T-1 keys) around median
fn split_child(ar:[]u8, nc:[]u8, T:i64, p:i64, i:i64)void {
    let y:i64=g_c(ar,p,i);
    let z:i64=node_new(ar,nc,g_leaf(ar,y));
    let j:i64=0;
    while(j<T-1){
        s_k(ar,z,j,g_k(ar,y,T+j));
        j=j+1;
    }
    if(g_leaf(ar,y)==0){
        let k:i64=0;
        while(k<T){
            s_c(ar,z,k,g_c(ar,y,T+k));
            k=k+1;
        }
    }
    s_n(ar,z,T-1);
    s_n(ar,y,T-1);
    // insert z as child i+1 of p
    let np:i64=g_n(ar,p);
    let c:i64=np;
    while(c>i){
        s_c(ar,p,c+1,g_c(ar,p,c));
        c=c-1;
    }
    s_c(ar,p,i+1,z);
    // insert median key y.keys[T-1] at position i
    let kk:i64=np-1;
    while(kk>=i){
        s_k(ar,p,kk+1,g_k(ar,p,kk));
        kk=kk-1;
    }
    s_k(ar,p,i,g_k(ar,y,T-1));
    s_n(ar,p,np+1);
    return;
}

        if(k<g_k(ar,nid,i2)){i2=i2-1;} else {break;}
    }
    i2=i2+1;
    let ch:i64=g_c(ar,nid,i2);
    if(g_n(ar,ch)==2*T-1){
        split_child(ar,nc,T,nid,i2);
        if(k>g_k(ar,nid,i2)){i2=i2+1;}
        ch=g_c(ar,nid,i2);
    }
    insert_nonfull(ar,nc,T,ch,k);
    return;
}

fn btree_insert(ar:[]u8, nc:[]u8, rt:[]u8, T:i64, k:i64)void {
    let r:i64=get64(rt,0);
    if(g_n(ar,r)==2*T-1){
        let s:i64=node_new(ar,nc,0);
        s_c(ar,s,0,r);
        put64(rt,0,s);
        split_child(ar,nc,T,s,0);
        insert_nonfull(ar,nc,T,s,k);
        return;
    }
    insert_nonfull(ar,nc,T,r,k);
    return;
}

fn bmerge(ar:[]u8, T:i64, p:i64, i:i64)void {
    let c:i64=g_c(ar,p,i);
    let s:i64=g_c(ar,p,i+1);
    let nc2:i64=g_n(ar,c);
    s_k(ar,c,nc2,g_k(ar,p,i));
    let j:i64=0;
    while(j<g_n(ar,s)){
        s_k(ar,c,nc2+1+j,g_k(ar,s,j));
        j=j+1;
    }
    if(g_leaf(ar,c)==0){
        let k:i64=0;
        while(k<=g_n(ar,s)){
            s_c(ar,c,nc2+1+k,g_c(ar,s,k));
            k=k+1;
        }
    }
    s_n(ar,c,nc2+1+g_n(ar,s));
    // mark s dead (flag at +88)
    put64(ar,n_off(s)+88,1);
    // remove key i and child i+1 from p
    let np:i64=g_n(ar,p);
    let k2:i64=i;
    while(k2<np-1){
        s_k(ar,p,k2,g_k(ar,p,k2+1));
        k2=k2+1;
    }
    let c2:i64=i+1;
    while(c2<np){
        s_c(ar,p,c2,g_c(ar,p,c2+1));
        c2=c2+1;
    }
    s_n(ar,p,np-1);
    return;
}

fn bfill(ar:[]u8, nc:[]u8, T:i64, p:i64, i:i64)i64 {
    let c:i64=g_c(ar,p,i);
    let np:i64=g_n(ar,p);
    // borrow from left sibling
    if(i>0){
        let sib:i64=g_c(ar,p,i-1);
        if(g_n(ar,sib)>=T){
            let ncb:i64=g_n(ar,c);
            // shift c keys/children right by 1
            let k:i64=ncb-1;
            while(k>=0){
                s_k(ar,c,k+1,g_k(ar,c,k));
                k=k-1;
            }
            if(g_leaf(ar,c)==0){
                let k2:i64=ncb;
                while(k2>=0){
                    s_c(ar,c,k2+1,g_c(ar,c,k2));
                    k2=k2-1;
                }
            }
            s_k(ar,c,0,g_k(ar,p,i-1));
            if(g_leaf(ar,c)==0){
                s_c(ar,c,0,g_c(ar,sib,g_n(ar,sib)));
            }
            s_k(ar,p,i-1,g_k(ar,sib,g_n(ar,sib)-1));
            s_n(ar,sib,g_n(ar,sib)-1);
            s_n(ar,c,ncb+1);
            return i;
        }
    }
    // borrow from right sibling
    if(i<np){
        let sib2:i64=g_c(ar,p,i+1);
        if(g_n(ar,sib2)>=T){
            let ncb2:i64=g_n(ar,c);
            s_k(ar,c,ncb2,g_k(ar,p,i));
            if(g_leaf(ar,c)==0){
                s_c(ar,c,ncb2+1,g_c(ar,sib2,0));
            }
            s_k(ar,p,i,g_k(ar,sib2,0));
            let ns2:i64=g_n(ar,sib2);
            let k3:i64=0;
            while(k3<ns2-1){
                s_k(ar,sib2,k3,g_k(ar,sib2,k3+1));
                k3=k3+1;
            }
            if(g_leaf(ar,sib2)==0){
                let k4:i64=0;
                while(k4<ns2){
                    s_c(ar,sib2,k4,g_c(ar,sib2,k4+1));
                    k4=k4+1;
                }
            }
            s_n(ar,sib2,ns2-1);
            s_n(ar,c,ncb2+1);
            return i;
        }
    }
    // merge
    if(i<np){
        bmerge(ar,T,p,i);
        return i;
    }
    bmerge(ar,T,p,i-1);
    return i-1;
}

fn bdelete(ar:[]u8, nc:[]u8, T:i64, nid:i64, k:i64)void {
    let n:i64=g_n(ar,nid);
    let i:i64=0;
    while(i<n){
        if(g_k(ar,nid,i)<k){i=i+1;} else {break;}
    }
    let found:i64=0;
    if(i<n){
        if(g_k(ar,nid,i)==k){found=1;}
    }
    if(found==1){
        if(g_leaf(ar,nid)==1){
            let j:i64=i;
            while(j<n-1){
                s_k(ar,nid,j,g_k(ar,nid,j+1));
                j=j+1;
            }
            s_n(ar,nid,n-1);
            return;
        }
        let c1:i64=g_c(ar,nid,i);
        let c2:i64=g_c(ar,nid,i+1);
        if(g_n(ar,c1)>=T){
            // predecessor
            let pc:i64=c1;
            while(g_leaf(ar,pc)==0){
                pc=g_c(ar,pc,g_n(ar,pc));
            }
            let pred:i64=g_k(ar,pc,g_n(ar,pc)-1);
            s_k(ar,nid,i,pred);
            bdelete(ar,nc,T,c1,pred);
            return;
        }
        if(g_n(ar,c2)>=T){
            // successor
            let sc:i64=c2;
            while(g_leaf(ar,sc)==0){
                sc=g_c(ar,sc,0);
            }
            let succ:i64=g_k(ar,sc,0);
            s_k(ar,nid,i,succ);
            bdelete(ar,nc,T,c2,succ);
            return;
        }
        bmerge(ar,T,nid,i);
        bdelete(ar,nc,T,c1,k);
        return;
    }
    if(g_leaf(ar,nid)==1){
        return;
    }
    let ch:i64=g_c(ar,nid,i);
    if(g_n(ar,ch)<T){
        i=bfill(ar,nc,T,nid,i);
        ch=g_c(ar,nid,i);
    }
    bdelete(ar,nc,T,ch,k);
    return;
}

fn btree_delete(ar:[]u8, nc:[]u8, rt:[]u8, T:i64, k:i64)void {
    let r:i64=get64(rt,0);
    bdelete(ar,nc,T,r,k);
    let r2:i64=get64(rt,0);
    if(g_leaf(ar,r2)==0){
        if(g_n(ar,r2)==0){
            put64(ar,n_off(r2)+88,1);
            put64(rt,0,g_c(ar,r2,0));
        }
    }
    return;
}

fn read_all(path:[]u8, mem:*u8, cap:i64)i64 {
    let pp:*u8=path_cstr(path);
    let fd:i64=_zag_raw_syscall(2, pp as i64, 0, 0, 0, 0, 0);
    if(fd<0){return -1;}
    let total:i64=0;
    while(1==1){
        if(total>=cap){break;}
        let want:i64=cap-total;
        if(want>65536){want=65536;}
        let r:i64=_zag_raw_syscall(0, fd, (mem as i64)+total, want, 0, 0, 0);
        if(r<=0){break;}
        total=total+r;
    }
    let c:i64=_zag_raw_syscall(3, fd, 0, 0, 0, 0, 0);
    return total;
}

fn s_eq(a:[]u8, b:[]u8)i64 {
    if(a.len!=b.len){return 0;}
    let i:i64=0;
    while(i<a.len){
        if(a[i]!=b[i]){return 0;}
        i=i+1;
    }
    return 1;
}

fn parse_i64(s:[]u8)i64 {
    let neg:i64=0;
    let i:i64=0;
    if(s.len>0){
        if(s[0]==45){neg=1;i=1;}
    }
    let v:i64=0;
    while(i<s.len){
        let c:u8=s[i];
        if(c>=48){
            if(c<=57){v=v*10+((c-48) as i64);}
        }
        i=i+1;
    }
    if(neg==1){v=0-v;}
    return v;
}

fn i64s(v:i64)[]u8 {
    let t:[]u8=_zag_i64_to_str(v);
    let n:i64=t.len;
