
fn b2_predict(prog:[]u8, nn:i32, inp:[]u8, out:[]u8)i32 {
  let key:[]u8="rab";
  let val:[]u8="rrr";
  let hit:i32=1;
  if(inp.len!=key.len){ hit=0; }
  else {
    let i:i32=0;
    while(i<inp.len){
      if(inp[i]!=key[i]){ hit=0; }
      i=i+1;
    }
  }
  if(hit==1){
    let j:i32=0;
    while(j<val.len){ out[j]=val[j]; j=j+1; }
    return val.len;
  }
  let n:i32=inp.len;
  let k:i32=0;
  while(k<n){
    let id:i32=eval_prog(prog, nn, k, n);
    out[k]=inp[id];
    k=k+1;
  }
  return n;
}

fn main()i32 {
  emit("=== B2 memorization control ===\n");
  let ps:[]u8=z_alloc(200000);
  let pidx:[]u8=z_alloc(1200*4);
  let pcount:i32=benum(ps, pidx);
  emit("enumerated ");
  emit(i32s(pcount));
  emit(" programs\n");
  let tmp:[]u8=z_alloc(1);
  let prog:[]u8=get_prog(ps, pidx, 38, tmp);
  let nn:i32=tmp[0] as i32;
  let out:[]u8=z_alloc(8);
  let n1:i32=b2_predict(prog, nn, "rab", out);
  emit("B2 PREDICT rab -> ");
  emit(out[0..n1]);
  emit("\n");
  let n2:i32=b2_predict(prog, nn, "rqw", out);
  emit("B2 PREDICT rqw -> ");
  emit(out[0..n2]);
  emit("\n");
  return 0;
}
