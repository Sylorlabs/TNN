
fn main()i32 {
  emit("=== B1 full re-enumeration over T+F1r ===\n");
  let ps:[]u8=z_alloc(200000);
  let pidx:[]u8=z_alloc(1200*4);
  let pcount:i32=benum(ps, pidx);
  emit("enumerated ");
  emit(i32s(pcount));
  emit(" programs\n");
  let sdat:[]u8=z_alloc(4*64);
  let sod:[]u8=z_alloc(16);
  let sld:[]u8=z_alloc(16);
  let snd:[]u8=z_alloc(16);
  let _l:i32=loadseq(sdat, sod, sld, snd, 0, "abc", "ccc", 3);
  let _l:i32=loadseq(sdat, sod, sld, snd, 1, "xy", "yy", 2);
  let _l:i32=loadseq(sdat, sod, sld, snd, 2, "defg", "gggg", 4);
  let _l:i32=loadseq(sdat, sod, sld, snd, 3, "rab", "rrr", 3);
  let tmp:[]u8=z_alloc(1);
  let lnb:[]u8=z_alloc(4);
  let evaluated:i32=0;
  let found:i32=-1;
  let pi:i32=0;
  while(pi<pcount){
    let prog:[]u8=get_prog(ps, pidx, pi, tmp);
    let nn:i32=tmp[0] as i32;
    evaluated=evaluated+1;
    let ok:i32=1;
    let s:i32=0;
    while(s<4){
      let sq:[]u8=getseq(sdat, sod, sld, s, lnb);
      let ln:i32=get32(lnb, 0);
      let n:i32=get32(snd, s*4);
      if(prog_fits(prog, nn, sq, ln, n)!=1){ ok=0; s=4; }
      s=s+1;
    }
    if(ok==1){ found=pi; pi=pcount; }
    pi=pi+1;
  }
  emit("B1 programs-evaluated: ");
  emit(i32s(evaluated));
  emit("\n");
  emit("B1 first-fit-index: ");
  emit(i32s(found));
  emit("\n");
  return 0;
}
