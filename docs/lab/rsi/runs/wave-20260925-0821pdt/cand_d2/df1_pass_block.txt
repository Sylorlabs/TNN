// ---------------- DF-1 foreground defocus (frozen, wave-20260925-0821pdt) ----------------
// Final lens pass. Frozen-radius (R=7) separable integer box blur applied only
// to the foreground tier below the plain horizon, with a frozen 12 px feather
// band and the arch opening carved out (the far rim seen through the arch is
// at infinity focus and stays sharp). The near field defocuses while giant,
// moon, sky, and focus plane stay bit-identical to the baseline. No RNG, no
// new samples, nothing read from outside the canvas. Deterministic by design.
fn r8c_df1_w12(x:i64, y:i64) i64 {
    let py:i64 = r8c_plain_y(x);
    if (r8c_arch_open(x, y) == 1) { return 0; }
    if (y <= py) { return 0; }
    if (y >= py + 12) { return 12; }
    return y - py;
}
fn r8c_pass6_defocus(buf:[]u8, stride:i64, w:i64, h:i64, tr:i64) void {
    // frozen parameters
    let R:i64 = 7;
    let K:i64 = 15;
    let F:i64 = 12;
    // mask: one u8 per pixel, w12 in 0..12. x-outer so r8c_plain_y is
    // evaluated once per column.
    let mask:[]u8 = nio_alloc((w * h) as i32);
    if (mask.len == 0) { _zag_println("df1 mask alloc failed"); return; }
    let blur_n:i64 = 0;
    let carve_n:i64 = 0;
    let mx:i64 = 0;
    while (mx < w) {
        let py:i64 = r8c_plain_y(mx);
        let my:i64 = 0;
        while (my < h) {
            let w12:i64 = 0;
            if (r8c_arch_open(mx, my) == 1) {
                if (my > py) { carve_n = carve_n + 1; }
            } else if (my > py) {
                if (my >= py + F) { w12 = F; } else { w12 = my - py; }
            }
            if (w12 > 0) { blur_n = blur_n + 1; }
            mask[my * w + mx] = w12 as u8;
            my = my + 1;
        }
        mx = mx + 1;
    }
    // scratch: horizontal pass output, same BGR layout as buf
    let tmp:[]u8 = nio_alloc((54 + h * stride) as i32);
    if (tmp.len == 0) { _zag_println("df1 scratch alloc failed"); nio_free(mask); return; }
    let hy:i64 = 0;
    while (hy < h) {
        let hx:i64 = 0;
        while (hx < w) {
            let sb:i64 = 0;
            let sg:i64 = 0;
            let sr:i64 = 0;
            let hk:i64 = 0 - R;
            while (hk <= R) {
                let xx:i64 = hx + hk;
                if (xx < 0) { xx = 0; }
                if (xx >= w) { xx = w - 1; }
                let o:i64 = 54 + hy * stride + xx * 3;
                sb = sb + (buf[o] as i64);
                sg = sg + (buf[o + 1] as i64);
                sr = sr + (buf[o + 2] as i64);
                hk = hk + 1;
            }
            let o2:i64 = 54 + hy * stride + hx * 3;
            tmp[o2] = ((sb + 7) / K) as u8;
            tmp[o2 + 1] = ((sg + 7) / K) as u8;
            tmp[o2 + 2] = ((sr + 7) / K) as u8;
            hx = hx + 1;
        }
        hy = hy + 1;
    }
    // vertical pass + feather blend. Pixels with w12 == 0 are never written.
    let vy:i64 = 0;
    while (vy < h) {
        let vx:i64 = 0;
        while (vx < w) {
            let wv:i64 = mask[vy * w + vx] as i64;
            if (wv > 0) {
                let bb:i64 = 0;
                let bg:i64 = 0;
                let br:i64 = 0;
                let vk:i64 = 0 - R;
                while (vk <= R) {
                    let yy:i64 = vy + vk;
                    if (yy < 0) { yy = 0; }
                    if (yy >= h) { yy = h - 1; }
                    let o:i64 = 54 + yy * stride + vx * 3;
                    bb = bb + (tmp[o] as i64);
                    bg = bg + (tmp[o + 1] as i64);
                    br = br + (tmp[o + 2] as i64);
                    vk = vk + 1;
                }
                bb = (bb + 7) / K;
                bg = (bg + 7) / K;
                br = (br + 7) / K;
                let o2:i64 = 54 + vy * stride + vx * 3;
                let s0:i64 = buf[o2] as i64;
                let s1:i64 = buf[o2 + 1] as i64;
                let s2:i64 = buf[o2 + 2] as i64;
                buf[o2] = ((s0 * (F - wv) + bb * wv + 6) / F) as u8;
                buf[o2 + 1] = ((s1 * (F - wv) + bg * wv + 6) / F) as u8;
                buf[o2 + 2] = ((s2 * (F - wv) + br * wv + 6) / F) as u8;
            }
            vx = vx + 1;
        }
        vy = vy + 1;
    }
    nio_free(mask);
    nio_free(tmp);
    r8c_tr_s(tr, "## DF-1 foreground defocus (frozen lens pass)\n");
    r8c_tr_s(tr, "DF1: separable integer box blur, radius R=7 (15x15), 12 px feather at the plain horizon.\n");
    r8c_tr_s(tr, "DF1: arch opening carved out (far rim at infinity focus stays sharp).\n");
    r8c_tr_s(tr, "DF1: blur-zone pixel count=");
    r8c_tr_n(tr, blur_n);
    r8c_tr_nl(tr);
    r8c_tr_s(tr, "DF1: carve-out pixel count=");
    r8c_tr_n(tr, carve_n);
    r8c_tr_nl(tr);
}
