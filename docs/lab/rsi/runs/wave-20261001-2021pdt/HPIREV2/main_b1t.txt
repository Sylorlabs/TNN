
// B1T main: the B1 discovery enumeration benum (1055 programs,
// byte-identical algorithm to the frozen v1 discovery) run over T
// alone (the set the frozen v1 discovery fit), taking the first program
// fitting all of T, then verifying that fitted program on T.
// Required by the amended step-5 prereg
// (PREREG_PI_REV2_STEP5_BASELINE_AMENDED.md, K-SB4b): B1 run over T
// alone must return first_fit >= 0 with fails=0 on T, pinning B1 as a
// working re-search baseline rather than a broken one.
// Shared code above this line is byte-exact extracts of the frozen
// discovery machinery from proc_revise2.zag (committed 847a8f10f).
// Pure Zag. Pinned znc.

fn main()i32 {
  emit("=== B1T full re-enumeration over T alone ===\n");
  let ps:[]u8=z_alloc(200000);
  let pidx:[]u8=z_alloc(1200*4);
  let pcount:i32=benum(ps, pidx);
  emit("enumerated ");
  emit(i32s(pcount));
  emit(" programs\n");
  let sdat:[]u8=z_alloc(4*64);
  let sod:[]u8=z_alloc(16);
  let sld:[]u8=z_alloc(16);
  let snd:[]u8=z_alloc(16);
  let _l:i32=loadseq(sdat, sod, sld, snd, 0, "abc", "ccc", 3);
  let _l:i32=loadseq(sdat, sod, sld, snd, 1, "xy", "yy", 2);
  let _l:i32=loadseq(sdat, sod, sld, snd, 2, "defg", "gggg", 4);
  let tmp:[]u8=z_alloc(1);
  let lnb:[]u8=z_alloc(4);
  let evaluated:i32=0;
  let found:i32=-1;
  let pi:i32=0;
  while(pi<pcount){
    let prog:[]u8=get_prog(ps, pidx, pi, tmp);
    let nn:i32=tmp[0] as i32;
    evaluated=evaluated+1;
    let ok:i32=1;
    let s:i32=0;
    while(s<3){
      let sq:[]u8=getseq(sdat, sod, sld, s, lnb);
      let ln:i32=get32(lnb, 0);
      let n:i32=get32(snd, s*4);
      if(prog_fits(prog, nn, sq, ln, n)!=1){ ok=0; s=3; }
      s=s+1;
    }
    if(ok==1){ found=pi; pi=pcount; }
    pi=pi+1;
  }
  emit("B1T programs-evaluated: ");
  emit(i32s(evaluated));
  emit("\n");
  emit("B1T first-fit-index: ");
  emit(i32s(found));
  emit("\n");
  let fails:i32=0;
  if(found>=0){
    let prog:[]u8=get_prog(ps, pidx, found, tmp);
    let nn:i32=tmp[0] as i32;
    let s:i32=0;
    while(s<3){
      let sq:[]u8=getseq(sdat, sod, sld, s, lnb);
      let ln:i32=get32(lnb, 0);
      let n:i32=get32(snd, s*4);
      if(prog_fits(prog, nn, sq, ln, n)!=1){ fails=fails+1; }
      s=s+1;
    }
  } else {
    fails=fails+1;
  }
  emit("B1T fails-on-T: ");
  emit(i32s(fails));
  emit("\n");
  return 0;
}
