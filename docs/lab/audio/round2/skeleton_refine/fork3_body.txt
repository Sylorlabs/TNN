// fork3 body — appended after the prelude.
const SEED:i64 = 1400200003;
const NF0:i64 = 8;

fn note_f0(k:i64) f64 {
    if (k == 0) { return 523.25; }
    if (k == 1) { return 587.33; }
    if (k == 2) { return 659.25; }
    if (k == 3) { return 783.99; }
    if (k == 4) { return 880.0; }
    if (k == 5) { return 783.99; }
    if (k == 6) { return 659.25; }
    return 587.33;
}

// hash-seeded micro-timing: note k onset shift in [-8ms, +8ms]
fn onset_shift(k:i64) f64 {
    let h:i64 = hash01(k * 7919 + SEED);
    return ((h as f64) / 65535.0 - 0.5) * 0.016;
}

fn render(mode_full:i64, acc:[]u8, nsamp:i64) void {
    let notedur:f64 = 0.45;
    let gap:f64 = 0.12;
    let step:f64 = notedur + gap;
    // phase-locked harmonic stack; breath state; transient state
    let blp:f64 = 0.0;
    let blp2:f64 = 0.0;
    let k:i64 = 0;
    while (k < NF0) {
        let f0n:f64 = note_f0(k);
        let t0:f64 = (k as f64) * step + onset_shift(k);
        if (t0 < 0.0) { t0 = 0.0; }
        let t1:f64 = t0 + notedur;
        let s0:i64 = (t0 * 44100.0) as i64;
        let s1:i64 = (t1 * 44100.0) as i64;
        if (s0 < 0) { s0 = 0; }
        if (s1 > nsamp) { s1 = nsamp; }
        let atk:i64 = 661;   // 15 ms
        let rel:i64 = 2646;  // 60 ms
        // fundamental phase accumulator (vibrato applied in F0)
        let ph:f64 = 0.0;
        let s:i64 = s0;
        while (s < s1) {
            let t:f64 = (s as f64) / 44100.0;
            let vib:f64 = 1.0 + 0.008 * sinr(6.283185307179586 * 5.5 * t + 1.0);
            let f0:f64 = f0n * vib;
            ph = ph + 6.283185307179586 * f0 / 44100.0;
            // per-note envelope (raised-cosine attack/release)
            let e:f64 = 1.0;
            let ds0:i64 = s - s0;
            if (ds0 < atk) {
                let u:f64 = (ds0 as f64) / (atk as f64);
                e = 0.5 - 0.5 * cosr(3.141592653589793 * u);
            }
            let ds1:i64 = s1 - s;
            if (ds1 < rel) {
                let u:f64 = (ds1 as f64) / (rel as f64);
                let er:f64 = 0.5 - 0.5 * cosr(3.141592653589793 * u);
                if (er < e) { e = er; }
            }
            let v:f64 = 7000.0 * sinr(ph);
            if (mode_full != 0) {
                // refinement: harmonics 2..20, 1/k, phase-locked, anti-aliased
                let kk:i64 = 2;
                while (kk <= 20) {
                    let hk:f64 = (kk as f64) * f0n * vib;
                    if (hk < 20000.0) {
                        v = v + (7000.0 / (kk as f64)) * sinr((kk as f64) * ph);
                    }
                    kk = kk + 1;
                }
                // breath: hash-indexed noise, one-pole LP (~4 kHz), -X dB
                let hb:i64 = hash01(s ^ 0x9e3779b97f4a7c15);
                let bn:f64 = ((hb as f64) / 65535.0 - 0.5) * 2.0;
                blp = blp + 0.25 * (bn - blp);
                blp2 = blp2 + 0.25 * (blp - blp2);
                v = v + 17000.0 * blp2;
                // onset transient: 3 ms shaped burst at note attack
                if (ds0 < 132) {
                    let u:f64 = (ds0 as f64) / 132.0;
                    let ht:i64 = hash01((s * 131 + 17) ^ 0x9e3779b97f4a7c15);
                    let tn:f64 = ((ht as f64) / 65535.0 - 0.5) * 2.0;
                    let shape:f64 = sinr(3.141592653589793 * u);
                    v = v + 1260.0 * tn * shape;
                }
            }
            v = v * e;
            put64(acc, s * 8, get64(acc, s * 8) + (v as i64));
            s = s + 1;
        }
        k = k + 1;
    }
}

fn main() i32 {
    let outpath:[]u8 = _zag_arg(1);
    let mode:[]u8 = _zag_arg(2);
    let full:i64 = 1;
    if (str_eq(mode, "skel") != 0) { full = 0; }
    let nsamp:i64 = 215000; // ~4.88 s
    let acc:[]u8 = nio_alloc((nsamp * 8) as i32);
    if (acc.len == 0) { pout("alloc fail\n"); return 1; }
    render(full, acc, nsamp);
    // fixed DC block (deterministic, not output-derived)
    let x1:i64 = 0; let y1:i64 = 0;
    let s:i64 = 0;
    while (s < nsamp) {
        let x0:i64 = get64(acc, s * 8);
        let y0:i64 = x0 - x1 + (y1 * 1023) / 1024;
        x1 = x0; y1 = y0;
        put64(acc, s * 8, y0);
        s = s + 1;
    }
    let rc:i64 = wav_write(outpath, acc, nsamp);
    pout("rc="); pout(i64s(rc)); pout(" mode_full="); pout(i64s(full)); pout("\n");
    nio_free(acc);
    return 0;
}
