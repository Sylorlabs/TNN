// fork4 body v2 — appended after the prelude.
// Source-filter: band-limited saw (harmonics 1..12, 1/k, phase-locked)
// × FROZEN 3-formant mask. A = static F0; B = + prosody layer.
const SEED:i64 = 1400200004;

fn note_f0(k:i64) f64 {
    if (k == 0) { return 440.0; }
    if (k == 1) { return 493.88; }
    if (k == 2) { return 523.25; }
    if (k == 3) { return 587.33; }
    if (k == 4) { return 659.25; }
    return 587.33;
}
fn onset_shift(k:i64) f64 {
    let h:i64 = hash01(k * 7919 + SEED);
    return ((h as f64) / 65535.0 - 0.5) * 0.016;
}
fn fgain(j:i64) f64 {
    if (j == 0) { return 1.0; }
    if (j == 1) { return 0.7; }
    return 0.5;
}
fn reso_init(F:f64, B:f64, c:[]u8, o:i64) void {
    let r:f64 = expf(-3.141592653589793 * B / 44100.0);
    let th:f64 = 6.283185307179586 * F / 44100.0;
    put64(c, o, ((1.0 - r * r) * 1000000.0) as i64);
    put64(c, o + 8, (2.0 * r * cosr(th) * 1000000.0) as i64);
    put64(c, o + 16, (r * r * 1000000.0) as i64);
}

fn render(with_prosody:i64, acc:[]u8, nsamp:i64) void {
    let c:[]u8 = nio_alloc(72 as i32);
    reso_init(800.0, 100.0, c, 0);
    reso_init(1200.0, 140.0, c, 24);
    reso_init(2800.0, 180.0, c, 48);
    let st:[]u8 = nio_alloc(48 as i32);
    let bs:[]u8 = nio_alloc(48 as i32); // breath resonator state
    let blp:f64 = 0.0; let blp2:f64 = 0.0;
    let notedur:f64 = 0.5; let step:f64 = 0.6;
    let k:i64 = 0;
    while (k < 6) {
        let f0n:f64 = note_f0(k);
        let t0:f64 = (k as f64) * step;
        if (with_prosody != 0) { t0 = t0 + onset_shift(k); }
        if (t0 < 0.0) { t0 = 0.0; }
        let s0:i64 = (t0 * 44100.0) as i64;
        let s1:i64 = ((t0 + notedur) * 44100.0) as i64;
        if (s0 < 0) { s0 = 0; }
        if (s1 > nsamp) { s1 = nsamp; }
        let atk:i64 = 661; let rel:i64 = 2646;
        let ph:f64 = 0.0;
        let n:i64 = s0;
        while (n < s1) {
            let t:f64 = (n as f64) / 44100.0;
            let vib:f64 = 1.0;
            if (with_prosody != 0) {
                vib = 1.0 + 0.009 * sinr(6.283185307179586 * 5.5 * t + 0.7);
            }
            let f0:f64 = f0n * vib;
            ph = ph + 6.283185307179586 * f0 / 44100.0;
            // band-limited saw source, phase-locked harmonics
            let src:f64 = sinr(ph);
            let kk:i64 = 2;
            while (kk <= 20) {
                let hk:f64 = (kk as f64) * f0;
                if (hk < 20000.0) { src = src + sinr((kk as f64) * ph) / (kk as f64); }
                kk = kk + 1;
            }
            src = src * 6000.0;
            // frozen formant mask
            let y:f64 = 0.0;
            let j:i64 = 0;
            while (j < 3) {
                let o:i64 = j * 24;
                let b0:f64 = (get64(c, o) as f64) / 1000000.0;
                let a1:f64 = (get64(c, o + 8) as f64) / 1000000.0;
                let a2:f64 = (get64(c, o + 16) as f64) / 1000000.0;
                let y1:f64 = (get64(st, j * 16) as f64) / 1000000.0;
                let y2:f64 = (get64(st, j * 16 + 8) as f64) / 1000000.0;
                let yn:f64 = b0 * src + a1 * y1 - a2 * y2;
                put64(st, j * 16, (yn * 1000000.0) as i64);
                put64(st, j * 16 + 8, (y1 * 1000000.0) as i64);
                y = y + fgain(j) * yn;
                j = j + 1;
            }
            let e:f64 = 1.0;
            let dn0:i64 = n - s0;
            if (dn0 < atk) {
                let u:f64 = (dn0 as f64) / (atk as f64);
                e = 0.5 - 0.5 * cosr(3.141592653589793 * u);
            }
            let dn1:i64 = s1 - n;
            if (dn1 < rel) {
                let u:f64 = (dn1 as f64) / (rel as f64);
                let er:f64 = 0.5 - 0.5 * cosr(3.141592653589793 * u);
                if (er < e) { e = er; }
            }
            let v:f64 = 0.5 * y * e;
            if (with_prosody != 0) {
                let hb:i64 = hash01(n ^ 0x9e3779b97f4a7c15);
                let bn:f64 = ((hb as f64) / 65535.0 - 0.5) * 2.0;
                // breath: single-pole LP'd (cutoff ~3.5 kHz) — tames the
                // white-noise HF that was blowing the rolloff gate, while
                // keeping enough broadband energy to move HNR.
                blp = blp + 0.45 * (bn - blp);
                v = v + 8840.0 * blp * e;
            }
            put64(acc, n * 8, get64(acc, n * 8) + (v as i64));
            n = n + 1;
        }
        k = k + 1;
    }
    nio_free(c); nio_free(st); nio_free(bs);
}

fn main() i32 {
    let outpath:[]u8 = _zag_arg(1);
    let mode:[]u8 = _zag_arg(2);
    let wp:i64 = 0;
    if (str_eq(mode, "B") != 0) { wp = 1; }
    let nsamp:i64 = 175000;
    let acc:[]u8 = nio_alloc((nsamp * 8) as i32);
    if (acc.len == 0) { pout("alloc fail\n"); return 1; }
    render(wp, acc, nsamp);
    let x1:i64 = 0; let y1:i64 = 0;
    let s:i64 = 0;
    while (s < nsamp) {
        let x0:i64 = get64(acc, s * 8);
        let y0:i64 = x0 - x1 + (y1 * 1023) / 1024;
        x1 = x0; y1 = y0;
        put64(acc, s * 8, y0);
        s = s + 1;
    }
    let rc:i64 = wav_write(outpath, acc, nsamp);
    pout("rc="); pout(i64s(rc)); pout(" prosody="); pout(i64s(wp)); pout("\n");
    nio_free(acc);
    return 0;
}
