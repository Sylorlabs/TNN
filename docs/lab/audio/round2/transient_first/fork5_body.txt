// fork5 body v4 — appended after the prelude.
// Clean integer-delay Karplus-Strong. Minimal, correct.
const SEED:i64 = 1400200005;

fn note_f0(k:i64) f64 {
    if (k == 0) { return 523.25; }
    if (k == 1) { return 587.33; }
    if (k == 2) { return 659.25; }
    if (k == 3) { return 783.99; }
    if (k == 4) { return 880.0; }
    if (k == 5) { return 783.99; }
    if (k == 6) { return 659.25; }
    return 587.33;
}

fn render(mode:i64, acc:[]u8, nsamp:i64) void {
    let damp:f64 = 0.996;
    if (mode == 2) { damp = 0.992; }
    let step:f64 = 1.0;
    let maxN:i64 = 120;
    let dl:[]u8 = nio_alloc((maxN * 8) as i32);
    let blp:f64 = 0.0; let blp2:f64 = 0.0; let blpb:f64 = 0.0;
    let k:i64 = 0;
    while (k < 8) {
        let f0n:f64 = note_f0(k);
        let N:i64 = (44100.0 / f0n) as i64;
        if (N > maxN) { N = maxN; }
        if (N < 2) { N = 2; }
        // init first N with dark noise
        let i:i64 = 0;
        let prev:f64 = 0.0;
        while (i < N) {
            let h:i64 = hash01(k * 100003 + i * 7919 + SEED);
            let v:f64 = ((h as f64) / 65535.0 - 0.5) * 2.0;
            prev = prev + 0.3 * (v - prev);
            put64(dl, i * 8, (prev * 1000000.0) as i64);
            i = i + 1;
        }
        let t0:f64 = (k as f64) * step;
        let s0:i64 = (t0 * 44100.0) as i64;
        let s1:i64 = ((t0 + 0.55) * 44100.0) as i64;
        if (s1 > nsamp) { s1 = nsamp; }
        let idx:i64 = 0;
        let n:i64 = s0;
        while (n < s1) {
            let dn:i64 = n - s0;
            // standard KS: out = dl[idx]; dl[idx] = damp*0.5*(dl[idx]+dl[idx+1])
            let cur:f64 = (get64(dl, idx * 8) as f64) / 1000000.0;
            let nxt:f64 = (get64(dl, ((idx + 1) % N) * 8) as f64) / 1000000.0;
            let upd:f64 = damp * 0.5 * (cur + nxt);
            put64(dl, idx * 8, (upd * 1000000.0) as i64);
            let v:f64 = 9000.0 * cur;
            if (mode != 1 && dn < 132) {
                let hb:i64 = hash01(n ^ 0x9e3779b97f4a7c15 ^ (k * 31));
                let bnv:f64 = ((hb as f64) / 65535.0 - 0.5) * 2.0;
                blpb = blpb + 0.4 * (bnv - blpb);
                v = v + 12000.0 * blpb;
            }
            let hb2:i64 = hash01(n ^ 0x5bd1e995);
            let bn2:f64 = ((hb2 as f64) / 65535.0 - 0.5) * 2.0;
            blp = blp + 0.25 * (bn2 - blp);
            blp2 = blp2 + 0.25 * (blp - blp2);
            v = v + 1000.0 * blp2;
            put64(acc, n * 8, get64(acc, n * 8) + (v as i64));
            idx = (idx + 1) % N;
            n = n + 1;
        }
        k = k + 1;
    }
    nio_free(dl);
}

fn main() i32 {
    let outpath:[]u8 = _zag_arg(1);
    let mode:[]u8 = _zag_arg(2);
    let m:i64 = 0;
    if (str_eq(mode, "noburst") != 0) { m = 1; }
    if (str_eq(mode, "fastdecay") != 0) { m = 2; }
    let nsamp:i64 = 385000;
    let acc:[]u8 = nio_alloc((nsamp * 8) as i32);
    if (acc.len == 0) { pout("alloc fail\n"); return 1; }
    render(m, acc, nsamp);
    let x1:i64 = 0; let y1:i64 = 0;
    let s:i64 = 0;
    while (s < nsamp) {
        let x0:i64 = get64(acc, s * 8);
        let y0:i64 = x0 - x1 + (y1 * 1023) / 1024;
        x1 = x0; y1 = y0;
        put64(acc, s * 8, y0);
        s = s + 1;
    }
    let rc:i64 = wav_write(outpath, acc, nsamp);
    pout("rc="); pout(i64s(rc)); pout(" mode="); pout(i64s(m)); pout("\n");
    nio_free(acc);
    return 0;
}
