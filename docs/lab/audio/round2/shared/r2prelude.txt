const SR:i64 = 44100;

// ---------- io ----------
fn cstr(s:[]u8) []u8 {
    if (s.len < 1 || s.len > 4095) { return ""; }
    let p:*u8 = _zag_malloc((s.len + 1) as i64) as *u8;
    if (p == null as *u8) { return ""; }
    let b:[]u8 = p[0..(s.len + 1)];
    let i:i64 = 0;
    while (i < s.len) { b[i] = s[i]; i = i + 1; }
    b[s.len] = 0;
    return b;
}
fn w_all(fd:i64, b:[]u8) i64 {
    let at:i64 = 0;
    let blen:i64 = b.len as i64;
    while (at < blen) {
        let n:i64 = _zag_raw_syscall(1, fd, (_zag_slice_ptr(b) as i64) + at, blen - at, 0, 0, 0);
        if (n == -4) { continue; }
        if (n <= 0) { return -7407; }
        at = at + n;
    }
    return at;
}
fn file_read(path:[]u8, cap:i64) []u8 {
    let cs:[]u8 = cstr(path);
    if (cs.len == 0) { return ""; }
    let fd:i64 = _zag_raw_syscall(2, (_zag_slice_ptr(cs) as i64), 0, 0, 0, 0, 0);
    _zag_free(_zag_slice_ptr(cs));
    if (fd < 0) { return ""; }
    let p:*u8 = _zag_malloc(cap) as *u8;
    if (p == null as *u8) { _zag_raw_syscall(3, fd, 0, 0, 0, 0, 0); return ""; }
    let at:i64 = 0;
    while (at < cap) {
        let n:i64 = _zag_raw_syscall(0, fd, (p as i64) + at, cap - at, 0, 0, 0);
        if (n <= 0) { break; }
        at = at + n;
    }
    _zag_raw_syscall(3, fd, 0, 0, 0, 0, 0);
    let b:[]u8 = p[0..at];
    return b;
}
fn file_write(path:[]u8, b:[]u8) i64 {
    let cs:[]u8 = cstr(path);
    if (cs.len == 0) { return -3; }
    let fd:i64 = _zag_raw_syscall(2, (_zag_slice_ptr(cs) as i64), 577, 438, 0, 0, 0);
    _zag_free(_zag_slice_ptr(cs));
    if (fd < 0) { return -4; }
    let wr:i64 = w_all(fd, b);
    _zag_raw_syscall(3, fd, 0, 0, 0, 0, 0);
    if (wr < 0) { return -5; }
    return 0;
}
fn pout(s:[]u8) void {
    _zag_raw_syscall(1, 1, (_zag_slice_ptr(s) as i64), (s.len as i64), 0, 0, 0);
}
fn i64s(v:i64) []u8 {
    let s:[]u8 = _zag_i64_to_str(v);
    if (s.len > 0) {
        if (s[s.len - 1] == 10) { return s[0..s.len - 1]; }
    }
    return s;
}
fn str_eq(a:[]u8, b:[]u8) i64 {
    if ((a.len as i64) != (b.len as i64)) { return 0; }
    let i:i64 = 0;
    while (i < a.len as i64) {
        if (a[i] != b[i]) { return 0; }
        i = i + 1;
    }
    return 1;
}

// ---------- arenas ([]u8 + explicit LE accessors; NEVER as []i32) ----------
fn nio_alloc(n:i32) []u8 {
    if (n < 1 || n > 33554432) { return ""; }
    let p:*u8 = _zag_malloc(n) as *u8;
    if (p == null as *u8) { return ""; }
    let b:[]u8 = p[0..n];
    let i:i32 = 0;
    while (i < n) { b[i] = 0; i = i + 1; }
    return b;
}
fn nio_free(b:[]u8) void { if (b.len > 0) { _zag_free(_zag_slice_ptr(b)); } }
fn put64(b:[]u8, i:i64, v:i64) void {
    b[i] = (v & 255) as u8;
    b[i+1] = ((v >> 8) & 255) as u8;
    b[i+2] = ((v >> 16) & 255) as u8;
    b[i+3] = ((v >> 24) & 255) as u8;
    b[i+4] = ((v >> 32) & 255) as u8;
    b[i+5] = ((v >> 40) & 255) as u8;
    b[i+6] = ((v >> 48) & 255) as u8;
    b[i+7] = ((v >> 56) & 255) as u8;
}
fn get64(b:[]u8, i:i64) i64 {
    return (b[i] as i64) | ((b[i+1] as i64) << 8) | ((b[i+2] as i64) << 16) | ((b[i+3] as i64) << 24) | ((b[i+4] as i64) << 32) | ((b[i+5] as i64) << 40) | ((b[i+6] as i64) << 48) | ((b[i+7] as i64) << 56);
}
fn get16s(b:[]u8, i:i64) i64 {
    let v:i64 = (b[i] as i64) | ((b[i+1] as i64) << 8);
    if (v >= 32768) { v = v - 65536; }
    return v;
}
fn put16(b:[]u8, i:i64, v:i64) void {
    let x:i64 = v;
    if (x < 0) { x = x + 65536; }
    b[i] = (x & 255) as u8;
    b[i+1] = ((x >> 8) & 255) as u8;
}

// ---------- deterministic hash (splitmix64, pure function — zero RNG) ----------
// Documented seed per fork. Used ONLY for micro-timing / dither.
fn hash64(x:i64) i64 {
    let z:i64 = x + (0x9e3779b97f4a7c15 as i64);
    z = (z ^ (z >> 30)) * (0xbf58476d1ce4e5b9 as i64);
    z = (z ^ (z >> 27)) * (0x94d049bb133111eb as i64);
    z = z ^ (z >> 31);
    return z;
}
fn hash01(x:i64) i64 {
    // returns Q16 in [0,65535] — pure function of x
    return (hash64(x) >> 16) & 65535;
}

// ---------- cos (Taylor, deterministic; corrected sign) ----------
fn fcosr(x:f64) f64 {
    let a:f64 = x;
    let sgn:f64 = 1.0;
    if (a > 1.5707963267948966) { a = 3.141592653589793 - a; sgn = -1.0; }
    let x2:f64 = a * a;
    let x4:f64 = x2 * x2;
    return sgn * (1.0 - x2 / 2.0 + x4 / 24.0 - x4 * x2 / 720.0 + x4 * x4 / 40320.0);
}
fn cosr(x:f64) f64 {
    let t:f64 = x / 6.283185307179586;
    let k:i64 = (t as i64);
    if (t < 0.0) { k = k - 1; }
    let r:f64 = x - (k as f64) * 6.283185307179586;
    // corrected: cos(r) = cos(2pi - r) for r in (pi, 2pi); no negation.
    if (r > 3.141592653589793) { return fcosr(6.283185307179586 - r); }
    return fcosr(r);
}
fn sinr(x:f64) f64 { return cosr(x - 1.5707963267948966); }

// ---------- wav out (FIXED gain staging; NO output-derived normalization) ----------
// acc: i64-per-sample arena. Caller guarantees |acc| < 32767 (peak < -1 dBFS
// must be proven by the analyzer on the delivered file). The clamp below is
// a safety net, never a mastering stage: hitting it fails the clip's gate.
fn wav_write(path:[]u8, acc:[]u8, nsamp:i64) i64 {
    let total:i64 = 44 + nsamp * 2;
    let out:[]u8 = nio_alloc(total as i32);
    if (out.len == 0) { return -11; }
    out[0] = 82; out[1] = 73; out[2] = 70; out[3] = 70;
    let sz:i64 = 36 + nsamp * 2;
    out[4] = (sz & 255) as u8; out[5] = ((sz >> 8) & 255) as u8;
    out[6] = ((sz >> 16) & 255) as u8; out[7] = ((sz >> 24) & 255) as u8;
    out[8] = 87; out[9] = 65; out[10] = 86; out[11] = 69;
    out[12] = 102; out[13] = 109; out[14] = 116; out[15] = 32;
    out[16] = 16; out[17] = 0; out[18] = 0; out[19] = 0;
    out[20] = 1; out[21] = 0;
    out[22] = 1; out[23] = 0;
    out[24] = 68; out[25] = 172; out[26] = 0; out[27] = 0;
    out[28] = 136; out[29] = 88; out[30] = 1; out[31] = 0;
    out[32] = 2; out[33] = 0;
    out[34] = 16; out[35] = 0;
    out[36] = 100; out[37] = 97; out[38] = 116; out[39] = 97;
    let ds:i64 = nsamp * 2;
    out[40] = (ds & 255) as u8; out[41] = ((ds >> 8) & 255) as u8;
    out[42] = ((ds >> 16) & 255) as u8; out[43] = ((ds >> 24) & 255) as u8;
    let i:i64 = 0;
    while (i < nsamp) {
        let v:i64 = get64(acc, i * 8);
        if (v > 32767) { v = 32767; }
        if (v < -32768) { v = -32768; }
        if (v < 0) { v = v + 65536; }
        let o:i64 = 44 + i * 2;
        out[o] = (v & 255) as u8;
        out[o+1] = ((v >> 8) & 255) as u8;
        i = i + 1;
    }
    let rc:i64 = file_write(path, out);
    nio_free(out);
    return rc;
}


// ---------- exp / ln (deterministic f64; for LF-model alpha + filter coeffs) ----------
fn exp_taylor(r:f64) f64 {
    let r2:f64 = r * r;
    let r3:f64 = r2 * r;
    let r4:f64 = r3 * r;
    let r5:f64 = r4 * r;
    let r6:f64 = r5 * r;
    return 1.0 + r + r2 / 2.0 + r3 / 6.0 + r4 / 24.0 + r5 / 120.0 + r6 / 720.0;
}
fn expf(x:f64) f64 {
    let k:i64 = ((x / 0.6931471805599453) + 0.5) as i64;
    if (x < 0.0) { k = ((x / 0.6931471805599453) - 0.5) as i64; }
    let r:f64 = x - (k as f64) * 0.6931471805599453;
    let e:f64 = exp_taylor(r);
    let p:f64 = 1.0;
    let n:i64 = k;
    if (n < 0) { n = -n; }
    let i:i64 = 0;
    while (i < n) { p = p * 2.0; i = i + 1; }
    if (k < 0) { return e / p; }
    return e * p;
}
fn lnf(x:f64) f64 {
    if (x <= 0.0) { return -1000000.0; }
    let y:f64 = (x - 1.0) / (x + 1.0);
    let y2:f64 = y * y;
    let t:f64 = y;
    let s:f64 = 0.0;
    let d:f64 = 1.0;
    let i:i64 = 0;
    while (i < 40) {
        s = s + t / d;
        t = t * y2;
        d = d + 2.0;
        i = i + 1;
    }
    return 2.0 * s;
}
