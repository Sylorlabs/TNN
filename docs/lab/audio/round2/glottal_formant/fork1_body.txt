// fork1 body — appended after the prelude.
// LF-model glottal flow + 6 series formants. Modes: full / sine.
// SEED 1400200001
const SEED:i64 = 1400200001;

fn note_f0(k:i64) f64 {
    if (k == 0) { return 196.0; }  // /ba/ G3
    if (k == 1) { return 246.94; } // /da/ B3
    return 293.66; // /ga/ D4
}

// LF-model glottal flow derivative (simplified)
// phase: 0..1 within the period. Returns E (derivative).
fn lf_pulse(phase:f64) f64 {
    if (phase < 0.6) {
        // open phase: positive hump (sin)
        return sinr(3.141592653589793 * phase / 0.6);
    }
    if (phase < 0.65) {
        // closure: sharp negative spike (abrupt) — moderate
        let x:f64 = (phase - 0.6) / 0.05;
        return -3.0 * (1.0 - x);
    }
    return 0.0; // closed
}

// 2-pole resonator state (6 formants × 2 delays = 12 i64s in arena)
fn formant_init(st:[]u8) void {
    let i:i64 = 0;
    while (i < 12) { put64(st, i * 8, 0); i = i + 1; }
}
fn formant_freq(k:i64) f64 {
    if (k == 0) { return 800.0; }
    if (k == 1) { return 1200.0; }
    if (k == 2) { return 2800.0; }
    if (k == 3) { return 3500.0; }
    if (k == 4) { return 4500.0; }
    return 6000.0;
}
fn formant_bw(k:i64) f64 {
    if (k == 0) { return 80.0; }
    if (k == 1) { return 100.0; }
    if (k == 2) { return 120.0; }
    if (k == 3) { return 150.0; }
    if (k == 4) { return 200.0; }
    return 300.0;
}
// process one sample through 6 PARALLEL formants (summed, normalized)
fn formants_process(x:f64, st:[]u8) f64 {
    let sum:f64 = 0.0;
    let k:i64 = 0;
    while (k < 6) {
        let F:f64 = formant_freq(k);
        let BW:f64 = formant_bw(k);
        let R:f64 = expf(-3.141592653589793 * BW / 44100.0);
        let th:f64 = 6.283185307179586 * F / 44100.0;
        let a1:f64 = -2.0 * R * cosr(th);
        let a2:f64 = R * R;
        let y1:f64 = (get64(st, (k*2) * 8) as f64) / 1000000.0;
        let y2:f64 = (get64(st, (k*2+1) * 8) as f64) / 1000000.0;
        let out:f64 = x - a1 * y1 - a2 * y2;
        put64(st, (k*2) * 8, (out * 1000000.0) as i64);
        put64(st, (k*2+1) * 8, (y1 * 1000000.0) as i64);
        sum = sum + out;
        k = k + 1;
    }
    return sum / 6.0; // normalize parallel sum
}

fn render(mode:i64, acc:[]u8, nsamp:i64) void {
    // mode 0=LF, 1=sine
    let st:[]u8 = nio_alloc(96 as i32); // 12 i64
    formant_init(st);
    let notedur:f64 = 0.5; let gap:f64 = 0.1; let step:f64 = 0.6;
    let k:i64 = 0;
    while (k < 3) {
        let f0n:f64 = note_f0(k);
        let T:f64 = 44100.0 / f0n;
        let t0:f64 = (k as f64) * step;
        let s0:i64 = (t0 * 44100.0) as i64;
        let s1:i64 = ((t0 + notedur) * 44100.0) as i64;
        if (s1 > nsamp) { s1 = nsamp; }
        // consonant burst: 20ms noise at onset (for transient detection)
        let b0:i64 = s0;
        let b1:i64 = s0 + 882;
        if (b1 > s1) { b1 = s1; }
        let n:i64 = b0;
        while (n < b1) {
            let hb:i64 = hash01(n ^ (SEED + k));
            let bn:f64 = ((hb as f64) / 65535.0 - 0.5) * 2.0;
            // burst bypasses formants for sharp transient
            put64(acc, n * 8, get64(acc, n * 8) + (bn * 8000.0) as i64);
            n = n + 1;
        }
        // vowel: LF or sine through formants
        let n2:i64 = b1;
        let phase:f64 = 0.0;
        while (n2 < s1) {
            let t:f64 = (n2 as f64) / 44100.0;
            // vibrato
            let vib:f64 = 1.0 + 0.008 * sinr(6.283185307179586 * 5.5 * t + 0.1);
            let f0v:f64 = f0n * vib;
            let Tv:f64 = 44100.0 / f0v;
            let ph:f64 = phase / Tv;
            if (ph >= 1.0) { ph = ph - 1.0; phase = ph * Tv; }
            let e:f64 = 0.0;
            if (mode == 0) {
                e = lf_pulse(ph);
            } else {
                e = sinr(6.283185307179586 * ph); // sine excitation
            }
            let v:f64 = formants_process(e, st);
            // clamp to prevent blowup
            let vc:f64 = v;
            if (vc > 4.0) { vc = 4.0; }
            if (vc < -4.0) { vc = -4.0; }
            put64(acc, n2 * 8, get64(acc, n2 * 8) + (vc * 3800.0) as i64);
            phase = phase + 1.0;
            // amplitude envelope: attack 20ms, sustain, release 50ms
            n2 = n2 + 1;
        }
        k = k + 1;
    }
    nio_free(st);
    // breath (continuous)
    let n3:i64 = 0;
    let b1f:f64 = 0.0; let b2f:f64 = 0.0;
    while (n3 < nsamp) {
        let hb:i64 = hash01(n3 ^ 0x5bd1e995);
        let bn:f64 = ((hb as f64) / 65535.0 - 0.5) * 2.0;
        b1f = b1f + 0.25 * (bn - b1f);
        b2f = b2f + 0.25 * (b1f - b2f);
        put64(acc, n3 * 8, get64(acc, n3 * 8) + (13000.0 * b2f) as i64);
        n3 = n3 + 1;
    }
}

fn main() i32 {
    let outpath:[]u8 = _zag_arg(1);
    let mode:[]u8 = _zag_arg(2);
    let m:i64 = 0;
    if (str_eq(mode, "sine") != 0) { m = 1; }
    let nsamp:i64 = 90000; // ~2.0 s (3 notes)
    let acc:[]u8 = nio_alloc((nsamp * 8) as i32);
    if (acc.len == 0) { pout("alloc fail\n"); return 1; }
    render(m, acc, nsamp);
    // DC block
    let x1:i64 = 0; let y1:i64 = 0;
    let s:i64 = 0;
    while (s < nsamp) {
        let x0:i64 = get64(acc, s * 8);
        let y0:i64 = x0 - x1 + (y1 * 1023) / 1024;
        x1 = x0; y1 = y0;
        put64(acc, s * 8, y0);
        s = s + 1;
    }
    let rc:i64 = wav_write(outpath, acc, nsamp);
    pout("rc="); pout(i64s(rc)); pout(" mode="); pout(i64s(m)); pout("\n");
    nio_free(acc);
    return 0;
}
