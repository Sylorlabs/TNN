// fork2 body v2 — appended after the prelude.
// True PSOLA: pitch-synchronous overlap of fossil periods.
// Modes: full / frozen / biggrain.
const SEED:i64 = 1400200002;

fn note_f0(k:i64) f64 {
    if (k == 0) { return 523.25; }
    if (k == 1) { return 587.33; }
    if (k == 2) { return 659.25; }
    if (k == 3) { return 783.99; }
    if (k == 4) { return 880.0; }
    if (k == 5) { return 783.99; }
    if (k == 6) { return 659.25; }
    return 587.33;
}
fn is_stressed(k:i64) i64 {
    if (k == 0) { return 1; }
    if (k == 3) { return 1; }
    if (k == 6) { return 1; }
    return 0;
}

fn fossil_load(path:[]u8, buf:[]u8, maxn:i64) i64 {
    let data:[]u8 = file_read(path, 16384);
    if (data.len < 2) { return 0; }
    let n:i64 = 0;
    let i:i64 = 0;
    while (i + 1 < (data.len as i64) && n < maxn) {
        let v:i64 = get16s(data, i);
        put64(buf, n * 8, v * 1000000);
        n = n + 1; i = i + 2;
    }
    return n;
}

fn render(mode:i64, acc:[]u8, nsamp:i64, fossil:[]u8, fn_samp:i64) void {
    // fossil PF=56. PSOLA: extract 2-period grains (112 samples),
    // resample to target period, overlap-add at target period.
    let pf:i64 = 56;
    let notedur:f64 = 0.6; let gap:f64 = 0.1; let step:f64 = 0.7;
    let blp:f64 = 0.0; let blp2:f64 = 0.0;
    // grain size: 32 periods (40ms, normal) or 128 periods (4x = 162ms, per prereg)
    let grain_periods:i64 = 32;
    if (mode == 2) { grain_periods = 128; }
    let k:i64 = 0;
    while (k < 8) {
        let f0n:f64 = note_f0(k);
        let P:i64 = (44100.0 / f0n) as i64; // target period
        if (P < 10) { P = 10; }
        if (P > 200) { P = 200; }
        let t0:f64 = (k as f64) * step;
        let s0:i64 = (t0 * 44100.0) as i64;
        let s1:i64 = ((t0 + notedur) * 44100.0) as i64;
        if (s1 > nsamp) { s1 = nsamp; }
        // PSOLA: overlap periods at P-sample intervals
        // grain = grain_periods * pf fossil samples, resampled to grain_periods * P
        let src_len:i64 = grain_periods * pf;
        let out_len:i64 = grain_periods * P;
        // fossil read position (cycles)
        let fpos:i64 = 0;
        let n:i64 = s0;
        while (n < s1) {
            // render one PSOLA grain centered at n
            let j:i64 = 0;
            while (j < out_len && n + j - out_len/2 >= 0 && n + j - out_len/2 < nsamp) {
                let out_idx:i64 = n + j - out_len / 2;
                if (out_idx >= s0 && out_idx < s1) {
                    // input position (resampled)
                    let inp:f64 = (j as f64) * (src_len as f64) / (out_len as f64);
                    let i0:i64 = (inp as i64);
                    let fr:f64 = inp - (i0 as f64);
                    let fp0:i64 = (fpos + i0) % fn_samp;
                    let fp1:i64 = (fpos + i0 + 1) % fn_samp;
                    let a:f64 = (get64(fossil, fp0 * 8) as f64) / 1000000.0;
                    let b:f64 = (get64(fossil, fp1 * 8) as f64) / 1000000.0;
                    let s:f64 = a * (1.0 - fr) + b * fr;
                    // Hanning over the grain
                    let w:f64 = 0.5 - 0.5 * cosr(6.283185307179586 * (j as f64) / (out_len as f64));
                    // vibrato REMOVED (fossil has natural variation; vibrato
                    // was pushing prosody over the 3% ceiling)
                    let vib:f64 = 1.0;
                    let v:f64 = 0.7 * s * w * vib;
                    put64(acc, out_idx * 8, get64(acc, out_idx * 8) + (v as i64));
                }
                j = j + 1;
            }
            if (mode != 1) {
                fpos = (fpos + pf) % fn_samp; // advance by one pitch period
            }
            // frozen (mode 1): fpos stays 0, single grain repeated
            // hop: P samples (one period), or P/2 if stressed (2x density)
            let hop:i64 = P;
            if (is_stressed(k) != 0 && mode == 0) { hop = P / 2; }
            n = n + hop;
        }
        k = k + 1;
    }
    // breath
    let n2:i64 = 0;
    let b1:f64 = 0.0; let b2:f64 = 0.0;
    while (n2 < nsamp) {
        let hb:i64 = hash01(n2 ^ 0x5bd1e995);
        let bn:f64 = ((hb as f64) / 65535.0 - 0.5) * 2.0;
        b1 = b1 + 0.25 * (bn - b1);
        b2 = b2 + 0.25 * (b1 - b2);
        put64(acc, n2 * 8, get64(acc, n2 * 8) + (15000.0 * b2) as i64);
        n2 = n2 + 1;
    }
}

fn main() i32 {
    let outpath:[]u8 = _zag_arg(1);
    let mode:[]u8 = _zag_arg(2);
    let m:i64 = 0;
    if (str_eq(mode, "frozen") != 0) { m = 1; }
    if (str_eq(mode, "biggrain") != 0) { m = 2; }
    let fossil:[]u8 = nio_alloc((8192 * 8) as i32);
    let fpath:[]u8 = _zag_arg(3);
    if (str_eq(fpath, "") != 0) { fpath = "fossil_hi.bin"; }
    let fn_samp:i64 = fossil_load(fpath, fossil, 8192);
    pout("fossil samples="); pout(i64s(fn_samp)); pout("\n");
    if (fn_samp == 0) { pout("fossil load fail\n"); return 1; }
    let nsamp:i64 = 270000;
    let acc:[]u8 = nio_alloc((nsamp * 8) as i32);
    if (acc.len == 0) { pout("alloc fail\n"); return 1; }
    render(m, acc, nsamp, fossil, fn_samp);
    let x1:i64 = 0; let y1:i64 = 0;
    let s:i64 = 0;
    while (s < nsamp) {
        let x0:i64 = get64(acc, s * 8);
        let y0:i64 = x0 - x1 + (y1 * 1023) / 1024;
        x1 = x0; y1 = y0;
        put64(acc, s * 8, y0);
        s = s + 1;
    }
    let rc:i64 = wav_write(outpath, acc, nsamp);
    pout("rc="); pout(i64s(rc)); pout(" mode="); pout(i64s(m)); pout("\n");
    nio_free(acc); nio_free(fossil);
    return 0;
}
