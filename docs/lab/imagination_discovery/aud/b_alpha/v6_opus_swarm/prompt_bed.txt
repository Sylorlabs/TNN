```zag
fn bed(mix:[]u8,nsamp:i64,txtab:[]u8,ntex:i64,txwave:[]u8,seed:i64,bgain:f64)void{
    if(ntex<=0){return;}
    // ... qualify textures: rms<600 && crest<7.0 -> ok[] ...
    let t:i64=0;
    let ti:i64=(h32(seed,9001) % nok);
    while(t<nsamp){
        let qi:i64=get64(ok,ti*8);          // pick texture qi
        let ln:i64=get32(txtab,qi*16);      // its length in samples
        let wo:i64=get64(txtab,q*16+8);     // its offset in the wave table
        // normalize chunk to common RMS 500:
        let norm:f64=500.0/(rms2+1.0);
        let wlen:i64=ln;
        let xf:i64=22050;                   // 0.5 s crossfade
        if(xf>wlen/2){xf=wlen/2;}
        let j:i64=0;
        while(j<wlen){
            let v:f64=((get16s(txwave,wo+j*2) as f64))/32768.0*bgain*norm;
            if(t>0 && j<xf){v=v*sstep((j as f64)/(xf as f64));}      // fade IN over first 0.5 s
            let rem:i64=wlen-j;
            if(rem<xf){v=v*sstep((rem as f64)/(xf as f64));}          // fade OUT over last 0.5 s
            // ... add into mix ...
        }
        t=t+(wlen-xf);                       // advance: next chunk overlaps by xf
        ti=(ti+1+(h32(seed,9002+t) % 3)) % nok;   // next texture (usually different)
    }
}
// sstep(x) = smoothstep: x*x*(3-2x), 0 for x<=0, 1 for x>=1
```
